# Tests are C executables; CTest does not require Python.
include(CTest)
if(BUILD_TESTING)
  file(GLOB regression_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tests/*.c")
  foreach(source IN LISTS regression_sources)
    get_filename_component(name "${source}" NAME_WE)
    add_executable(test_${name} "${source}")
    if(MSVC)
      target_compile_options(test_${name} PRIVATE /UNDEBUG)
    else()
      target_compile_options(test_${name} PRIVATE -UNDEBUG)
      target_link_libraries(test_${name} PRIVATE m)
    endif()
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/test-work/${name}")
    add_test(NAME ${name} COMMAND test_${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 10 WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/test-work/${name}")
  endforeach()
  add_test(NAME matrix_output COMMAND project1_moltiplicazione_matrici)
  set_tests_properties(matrix_output PROPERTIES PASS_REGULAR_EXPRESSION "58 64.*139 154" TIMEOUT 5)
endif()
